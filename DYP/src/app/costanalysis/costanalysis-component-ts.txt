import { Component, OnInit } from '@angular/core';
import {AngularFirestore} from '@angular/fire/firestore';
import {User} from '../model/user';
import {CalculationsService} from '../services/calculations.service';

@Component({
  selector: 'app-cost-analysis',
  templateUrl: './costanalysis.component.html',
  styleUrls: ['./costanalysis.component.css']
})
export class CostanalysisComponent implements OnInit {

  constructor(private db: AngularFirestore,
              private calculationsservice: CalculationsService) { }

  ngOnInit() {
    this.db.collection('users').snapshotChanges().subscribe(snaps => {
      const courses: User[] = snaps.map(snap => {
        return <User> {
          id: snap.payload.doc.id,
          ...snap.payload.doc.data()
        };
      });
      console.log(courses);
    });
    console.log('calculationsservice width' + this.calculationsservice.width);
    console.log('calculationsservice length' + this.calculationsservice.length);
    console.log('calculationsservice  area: ' + this.calculationsservice.calculateArea());

    console.log('calculationsservice weight' + this.calculationsservice.weight);
    console.log('calculationsservice scrapAmount' + this.calculationsservice.scrapamount);
    console.log('calculationsservice  grossWeight: ' + this.calculationsservice.calculateGrossWeight());
  }
}
